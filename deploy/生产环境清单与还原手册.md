### 短剧原生 2K 正式交付：固定提交发布、预检与回滚

该能力由 `huangque-content` 读取受控 `CONTENT_OUT` 中的 MiniMax 原生 2K 镜头，

合并并通过门禁的精确提交，禁止从浮动分支或服务器临时文件发布：

```bash
set -euo pipefail
HQ_RELEASE_COMMIT=<已合并且通过门禁的完整提交 SHA>
git fetch origin main
git checkout --detach "$HQ_RELEASE_COMMIT"
test "$(git rev-parse HEAD)" = "$HQ_RELEASE_COMMIT"
git merge-base --is-ancestor "$HQ_RELEASE_COMMIT" origin/main

# 此版本化清单是发布、备份和回滚的唯一边界；每行都是“仓库源文件<TAB>线上目标文件”。
# 禁止把 content_domains/、providers/ 或 workbench/ 整目录加入发布包。
HQ_RELEASE_MANIFEST=deploy/formal-delivery-release-manifest.tsv
test -f "$HQ_RELEASE_MANIFEST"
while IFS=$'\t' read -r HQ_SOURCE HQ_TARGET; do
  test -n "$HQ_SOURCE" && test -n "$HQ_TARGET"
  test -f "$HQ_SOURCE"
  case "$HQ_SOURCE:$HQ_TARGET" in
    */:*) exit 1 ;;
    server/*:/home/ubuntu/content-api/*) ;;
    site/workbench/*:/var/www/huangquechuanmei/workbench/*) ;;
    deploy/systemd/*:/etc/systemd/system/huangque-content.service.d/formal-delivery.conf) ;;
    *) exit 1 ;;
  esac
done <"$HQ_RELEASE_MANIFEST"

# 这些文件是导入依赖，不属于本 PR 的发布清单。仅校验存在、线上版本 SHA 和组合导入，
# 不复制到 live 路径。若 SHA 不同，应先按主站正常发布流程补齐依赖版本，再发布本能力。
for HQ_DEPENDENCY_SOURCE in \
  server/func_names.py \
  server/inspiration_cases.py \
  server/tikhub.py \
  server/providers/__init__.py \
  server/providers/short_drama_visual/__init__.py \
  server/providers/short_drama_visual/runtime.py
do
  test -f "$HQ_DEPENDENCY_SOURCE"
done

# 本段始终在本地固定 checkout 执行。先记录三个静态文件的预期 SHA，供远端磁盘和
# HTTPS 实际响应自动比较，不能只把两边 SHA 打印出来人工查看。
HQ_EXPECT_JS_SHA="$(sha256sum site/workbench/short-drama-workspace.js | awk '{print $1}')"
HQ_EXPECT_CSS_SHA="$(sha256sum site/workbench/short-drama-workspace.css | awk '{print $1}')"
HQ_EXPECT_HTML_SHA="$(sha256sum site/workbench/short-drama.html | awk '{print $1}')"
HQ_EXPECT_TRANSITION_SHA="$(sha256sum deploy/formal-delivery-live-transition.sh | awk '{print $1}')"

# 从显式清单构建精确发布包和校验文件。依赖 SHA 的路径直接写成线上绝对路径。
HQ_RELEASE_ARCHIVE="$(mktemp --suffix=.tgz)"
HQ_RELEASE_SHA256="$(mktemp)"
HQ_DEPENDENCY_SHA256="$(mktemp)"
while IFS=$'\t' read -r HQ_SOURCE HQ_TARGET; do
  sha256sum "$HQ_SOURCE" >>"$HQ_RELEASE_SHA256"
done <"$HQ_RELEASE_MANIFEST"
tar -czf "$HQ_RELEASE_ARCHIVE" -T <(cut -f1 "$HQ_RELEASE_MANIFEST")
for HQ_DEPENDENCY_MAPPING in \
  'server/func_names.py:/home/ubuntu/content-api/func_names.py' \
  'server/inspiration_cases.py:/home/ubuntu/content-api/inspiration_cases.py' \
  'server/tikhub.py:/home/ubuntu/content-api/tikhub.py' \
  'server/providers/__init__.py:/home/ubuntu/content-api/providers/__init__.py' \
  'server/providers/short_drama_visual/__init__.py:/home/ubuntu/content-api/providers/short_drama_visual/__init__.py' \
  'server/providers/short_drama_visual/runtime.py:/home/ubuntu/content-api/providers/short_drama_visual/runtime.py'
do
  HQ_DEPENDENCY_SOURCE="${HQ_DEPENDENCY_MAPPING%%:*}"
  HQ_DEPENDENCY_TARGET="${HQ_DEPENDENCY_MAPPING#*:}"
  printf '%s  %s\n' "$(sha256sum "$HQ_DEPENDENCY_SOURCE" | awk '{print $1}')" \
    "$HQ_DEPENDENCY_TARGET" >>"$HQ_DEPENDENCY_SHA256"
done

HQ_RELEASE_BACKUP="/home/ubuntu/release-backup/formal-delivery-$(date +%Y%m%d%H%M%S)"
HQ_RELEASE_STAGE="/home/ubuntu/release-staging/formal-delivery-$HQ_RELEASE_COMMIT"

# live payload 仍只来自精确清单；另行上传固定 checkout 中的控制脚本并校验独立 SHA。
# 两者都只进入版本化 staging，此时不得写 live 目录。
ssh dapeng-server "rm -rf '$HQ_RELEASE_STAGE' && mkdir -p '$HQ_RELEASE_STAGE/files'"
rsync -az "$HQ_RELEASE_ARCHIVE" \
  "dapeng-server:$HQ_RELEASE_STAGE/release.tgz"
rsync -az "$HQ_RELEASE_MANIFEST" \
  "dapeng-server:$HQ_RELEASE_STAGE/release-manifest.tsv"

  "dapeng-server:$HQ_RELEASE_STAGE/live-transition.sh"
ssh dapeng-server "tar -C '$HQ_RELEASE_STAGE/files' -xzf '$HQ_RELEASE_STAGE/release.tgz'"
rm -f "$HQ_RELEASE_ARCHIVE" "$HQ_RELEASE_SHA256" "$HQ_DEPENDENCY_SHA256"
```

随后仍在本地执行下面这个单一 SSH 门禁。远端脚本启用 `set -euo pipefail`；FFmpeg、

drop-in 状态为 `absent` 时会受控删除，而不是把新开关遗留在线上：
